let produtos = []; function validarProduto(descricao, quantidade, valor) { if (descricao.length < 5) { throw new Error("Descrição deve ter, no mínimo, cinco caracteres"); } if (quantidade < 1) { throw new Error("Quantidade deve ser maior que zero"); } if (valor < 0) { throw new Error("Valor deve ser maior ou igual a zero"); } } function cadastrarProduto(descricao, quantidade, valor) { validarProduto(descricao, quantidade, valor); let novoProduto = { codigo: produtos.length + 1, descricao: descricao, quantidade: quantidade, valor: valor }; produtos.push(novoProduto); } function atualizarValor(codigoProduto, novoValor) { if (novoValor < 0) { throw new Error("Valor deve ser maior ou igual a zero"); } const produto = produtos.find((prod) => prod.codigo === codigoProduto); if (produto) { produto.valor = novoValor; } else { throw new Error("Produto não encontrado"); } } function atualizarQuantidade(codigoProduto, novaQuantidade) { if (novaQuantidade < 1) { throw new Error("Quantidade deve ser maior que zero"); } const produto = produtos.find((prod) => prod.codigo === codigoProduto); if (produto) { produto.quantidade += novaQuantidade; } else { throw new Error("Produto não encontrado"); } } function listarProdutos() { console.log("Lista de produtos:"); produtos.forEach((p) => { console.log(`Código: ${p.codigo}, Descrição: ${p.descricao}, Quantidade: ${p.quantidade}, Valor: R$${p.valor}`); }); } // ============================ // TESTE E EXECUÇÃO DO SISTEMA // ============================ try { console.log("--- 1. cadastrando produtos ---"); cadastrarProduto("cadeira gamer", 12, 699); cadastrarProduto("mouse logitech", 38, 99); listarProdutos(); console.log("--- 2. atualizando valor ---"); atualizarValor(2, 97); listarProdutos(); console.log("--- 3. atualizando quantidade ---"); atualizarQuantidade(1, 3); listarProdutos(); } catch (erro) { console.error("Erro capturado:", erro.message); }
