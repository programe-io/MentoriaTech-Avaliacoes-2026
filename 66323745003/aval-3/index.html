print("===== CONTROLE DE IMC =====") nome = input("Digite o nome do paciente: ") peso = float(input("Digite o peso (kg): ")) altura = float(input("Digite a altura (m): ")) imc = peso / (altura ** 2) print("\n===== RESULTADO =====") print("Paciente:", nome) print("Peso:", peso, "kg") print("Altura:", altura, "m") print("IMC:", round(imc, 2)) if imc < 18.5: classificacao = "Abaixo do peso" elif imc < 25: classificacao = "Peso normal" elif imc < 30: classificacao = "Sobrepeso" elif imc < 35: classificacao = "Obesidade grau I" elif imc < 40: classificacao = "Obesidade grau II" else: classificacao = "Obesidade grau III" print("Classificação:", classificacao)
