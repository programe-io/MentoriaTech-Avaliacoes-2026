// Sistema de Validação Simples function validarNome(nome) { if (nome.trim() === "") { return "Erro: O nome não pode ficar vazio!"; } if (nome.length < 3) { return "Erro: O nome deve ter pelo menos 3 caracteres!"; } return "Nome válido!"; } function validarEmail(email) { const formatoEmail = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; if (email.trim() === "") { return "Erro: O e-mail não pode ficar vazio!"; } if (!formatoEmail.test(email)) { return "Erro: E-mail inválido!"; } return "E-mail válido!"; } function validarIdade(idade) { if (idade === "" || isNaN(idade)) { return "Erro: Informe uma idade válida!"; } idade = Number(idade); if (idade < 18 || idade > 120) { return "Erro: A idade deve estar entre 18 e 120 anos!"; } return "Idade válida!"; } function validarSenha(senha) { if (senha.length < 6) { return "Erro: A senha deve ter pelo menos 6 caracteres!"; } return "Senha válida!"; } // Testando as validações console.log("=== VALIDAÇÃO DE DADOS ==="); console.log(validarNome("Maria Silva")); console.log(validarEmail("maria@email.com")); console.log(validarIdade(25)); console.log(validarSenha("123456"));
