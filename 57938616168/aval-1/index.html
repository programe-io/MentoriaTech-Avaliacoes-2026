// 1. Dados dos Posts (Simulação de Banco de Dados)
const posts = [
    {
        id: 1,
        categoria: "Pele",
        titulo: "Como preparar a pele para uma maquiagem natural",
        resumo: "Limpeza, hidratação e protetor solar com cor são os 3 passos básicos para o dia a dia.",
        curtidas: 12
    },
    {
        id: 2,
        categoria: "Olhos",
        titulo: "Máscara de cílios sem empelotar",
        resumo: "Retire o excesso de produto do aplicador e faça movimentos em ziguezague da raiz às pontas.",
        curtidas: 24
    },
    {
        id: 3,
        categoria: "Boca",
        titulo: "Batom hidratante vs. Batom Matte",
        resumo: "Descubra qual opção escolher dependendo da ocasião e da hidratação dos seus lábios.",
        curtidas: 8
    }
];

// 2. Função para Exibir os Posts na Tela
function renderizarPosts(lista) {
    const container = document.getElementById("posts-container");
    container.innerHTML = "";

    lista.forEach(post => {
        container.innerHTML += `
            

                ${post.categoria}
                
${post.titulo}

                
${post.resumo}

                
            

        `;
    });
}

// 3. Função para Curtir um Post
function curtirPost(id) {
    const post = posts.find(p => p.id === id);
    if (post) {
        post.curtidas++;
        renderizarPosts(posts);
    }
}

// 4. Função para Filtrar Categoria
function filterPosts(categoria) {
    if (categoria === 'Todos') {
        renderizarPosts(posts);
    } else {
        const filtrados = posts.filter(p => p.categoria === categoria);
        renderizarPosts(filtrados);
    }
}

// Inicializar ao carregar a página
renderizarPosts(posts);
