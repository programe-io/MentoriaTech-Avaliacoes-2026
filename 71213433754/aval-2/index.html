// Dados
let tarefas = JSON.parse(
    localStorage.getItem("tarefas") || "[]"
);

let filtroAtual = "todas";

// Salvar tarefas
function salvar() {
    localStorage.setItem(
        "tarefas",
        JSON.stringify(tarefas)
    );
}

// Renderizar tarefas
function renderizar() {
    const lista = document.getElementById("lista");
    const contador = document.getElementById("contador");

    lista.innerHTML = "";

    let filtradas = tarefas.filter(tarefa => {
        if (filtroAtual === "pendentes")
            return !tarefa.concluida;

        if (filtroAtual === "concluidas")
            return tarefa.concluida;

        return true;
    });

    filtradas.forEach(tarefa => {
        const li = document.createElement("li");

        const checkbox = document.createElement("input");
        checkbox.type = "checkbox";
        checkbox.checked = tarefa.concluida;

        checkbox.addEventListener("change", () => {
            tarefa.concluida = checkbox.checked;
            salvar();
            renderizar();
        });

        const texto = document.createElement("span");
        texto.textContent = tarefa.nome;

        if (tarefa.concluida) {
            texto.classList.add("concluida");
        }

        const excluir = document.createElement("button");
        excluir.textContent = "Excluir";
        excluir.className = "excluir";

        excluir.addEventListener("click", () => {
            tarefas = tarefas.filter(t => t.id !== tarefa.id);
            salvar();
            renderizar();
        });

        li.append(checkbox, texto, excluir);
        lista.appendChild(li);
    });

    const pendentes = tarefas.filter(t => !t.concluida).length;
    const concluidas = tarefas.filter(t => t.concluida).length;

    contador.textContent =
        `Total: ${tarefas.length} | Pendentes: ${pendentes} | Concluídas: ${concluidas}`;
}

// Adicionar tarefa
function adicionarTarefa() {
    const input = document.getElementById("tarefaInput");
    const nome = input.value.trim();

    if (!nome) {
        alert("Digite uma tarefa!");
        return;
    }

    tarefas.push({
        id: Date.now(),
        nome: nome,
        concluida: false
    });

    input.value = "";

    salvar();
    renderizar();
}

// Botão adicionar
document.getElementById("adicionar")
    .addEventListener("click", adicionarTarefa);

// Adicionar com Enter
document.getElementById("tarefaInput")
    .addEventListener("keydown", evento => {
        if (evento.key === "Enter") {
            adicionarTarefa();
        }
    });

// Filtros
document.querySelectorAll("[data-filtro]")
    .forEach(botao => {
        botao.addEventListener("click", () => {
            filtroAtual = botao.dataset.filtro;

            document.querySelectorAll("[data-filtro]")
                .forEach(b => b.classList.remove("ativo"));

            botao.classList.add("ativo");

            renderizar();
        });
    });

// Inicializar
renderizar();
