let tarefas = []; // 1 - CADASTRAR TAREFA function cadastrarTarefa() { let codigo = Number(prompt("Digite o código da tarefa:")); // Verificar se o código já existe for (let i = 0; i < tarefas.length; i++) { if (tarefas[i].codigo === codigo) { alert("Código já cadastrado!"); return; } } let titulo = prompt("Digite o título da tarefa:"); if (titulo === null || titulo.trim().length < 5) { alert("O título precisa ter no mínimo 5 caracteres!"); return; } let prioridade = Number(prompt( "Digite a prioridade:\n1 - Alta\n2 - Média\n3 - Baixa" )); if (prioridade < 1 || prioridade > 3 || !Number.isInteger(prioridade)) { alert("Prioridade inválida!"); return; } let tarefa = { codigo: codigo, titulo: titulo.trim(), prioridade: prioridade, concluida: false }; tarefas.push(tarefa); alert("Tarefa cadastrada com sucesso!"); } // 2 - LISTAR TAREFAS function listarTarefas() { if (tarefas.length === 0) { alert("Nenhuma tarefa cadastrada!"); return; } let mensagem = ""; for (let i = 0; i < tarefas.length; i++) { mensagem += "Código: " + tarefas[i].codigo + "\nTítulo: " + tarefas[i].titulo + "\nPrioridade: " + tarefas[i].prioridade + "\nStatus: " + (tarefas[i].concluida ? "Concluída" : "Pendente") + "\n----------------------\n"; } alert(mensagem); } // 3 - CONCLUIR TAREFA function concluirTarefa() { let codigo = Number(prompt("Digite o código da tarefa:")); for (let i = 0; i < tarefas.length; i++) { if (tarefas[i].codigo === codigo) { tarefas[i].concluida = true; alert("Tarefa concluída!"); return; } } alert("Tarefa não encontrada!"); } // 4 - ALTERAR PRIORIDADE function alterarPrioridade() { let codigo = Number(prompt("Digite o código da tarefa:")); for (let i = 0; i < tarefas.length; i++) { if (tarefas[i].codigo === codigo) { let novaPrioridade = Number(prompt( "Digite a nova prioridade:\n1 - Alta\n2 - Média\n3 - Baixa" )); if (novaPrioridade < 1 || novaPrioridade > 3 || !Number.isInteger(novaPrioridade)) { alert("Prioridade inválida!"); return; } tarefas[i].prioridade = novaPrioridade; alert("Prioridade alterada com sucesso!"); return; } } alert("Tarefa não encontrada!"); } // 5 - MENU PRINCIPAL let opcao; do { opcao = prompt( "GERENCIAMENTO DE TAREFAS\n\n" + "1 - Cadastrar tarefa\n" + "2 - Listar tarefas\n" + "3 - Concluir tarefa\n" + "4 - Alterar prioridade\n" + "0 - Sair" ); switch (opcao) { case "1": cadastrarTarefa(); break; case "2": listarTarefas(); break; case "3": concluirTarefa(); break; case "4": alterarPrioridade(); break; case "0": alert("Programa encerrado!"); break; default: alert("Opção inválida!"); } } while (opcao !== "0");
