let pacientes = [];

function cadastrarPaciente(nome, idade, sexo, telefone) {
    let paciente = {
        id: Date.now(),
        nome: nome,
        idade: idade,
        sexo: sexo,
        telefone: telefone,
        alergias: [],
        doencas: [],
        medicamentos: [],
        sintomas: "",
        observacoes: ""
    };

    pacientes.push(paciente);
    console.log("Paciente cadastrado com sucesso!");
}

function adicionarAlergia(id, alergia) {
    let paciente = pacientes.find(p => p.id === id);

    if (paciente) {
        paciente.alergias.push(alergia);
        console.log("Alergia adicionada!");
    } else {
        console.log("Paciente não encontrado!");
    }
}

function adicionarDoenca(id, doenca) {
    let paciente = pacientes.find(p => p.id === id);

    if (paciente) {
        paciente.doencas.push(doenca);
        console.log("Doença adicionada!");
    }
}

function adicionarMedicamento(id, medicamento) {
    let paciente = pacientes.find(p => p.id === id);

    if (paciente) {
        paciente.medicamentos.push(medicamento);
        console.log("Medicamento registrado!");
    }
}

function registrarSintomas(id, sintomas) {
    let paciente = pacientes.find(p => p.id === id);

    if (paciente) {
        paciente.sintomas = sintomas;
        console.log("Sintomas registrados!");
    }
}

function adicionarObservacao(id, observacao) {
    let paciente = pacientes.find(p => p.id === id);

    if (paciente) {
        paciente.observacoes = observacao;
        console.log("Observação adicionada!");
    }
}

function listarPacientes() {
    console.log("\n===== PACIENTES CADASTRADOS =====");

    pacientes.forEach(p => {
        console.log(`ID: ${p.id}`);
        console.log(`Nome: ${p.nome}`);
        console.log(`Idade: ${p.idade}`);
        console.log(`Sexo: ${p.sexo}`);
        console.log(`Telefone: ${p.telefone}`);
        console.log("-----------------------------");
    });
}

function consultarAnamnese(id) {
    let p = pacientes.find(p => p.id === id);

    if (p) {
        console.log("\n===== FICHA DE ANAMNESE =====");
        console.log("Nome:", p.nome);
        console.log("Idade:", p.idade);
        console.log("Sexo:", p.sexo);
        console.log("Telefone:", p.telefone);
        console.log("Alergias:", p.alergias);
        console.log("Doenças:", p.doencas);
        console.log("Medicamentos:", p.medicamentos);
        console.log("Sintomas:", p.sintomas);
        console.log("Observações:", p.observacoes);
    } else {
        console.log("Paciente não encontrado!");
    }
}

function removerPaciente(id) {
    pacientes = pacientes.filter(p => p.id !== id);
    console.log("Paciente removido!");
}

// TESTANDO O SISTEMA

cadastrarPaciente("João Silva", 30, "Masculino", "99999-9999");

let idPaciente = pacientes[0].id;

adicionarAlergia(idPaciente, "Dipirona");
adicionarDoenca(idPaciente, "Asma");
adicionarMedicamento(idPaciente, "Bombinha");
registrarSintomas(idPaciente, "Dor de cabeça e tosse");
adicionarObservacao(idPaciente, "Paciente relata sintomas há três dias.");

listarPacientes();
consultarAnamnese(idPaciente);
