import random import time print("=" * 40) print(" 🔥 MINI BATTLE ROYALE 🔥") print("=" * 40) nome = input("Digite seu nome de jogador: ") vida = 100 municao = 30 kills = 0 inimigos = 10 armas = ["Pistola", "AK47", "MP40", "M1014"] arma = random.choice(armas) print(f"\n🎮 Jogador: {nome}") print(f"🔫 Arma inicial: {arma}") print("🪂 Você caiu na ilha!") while vida > 0 and inimigos > 0: print("\n" + "=" * 30) print(f"❤️ Vida: {vida}") print(f"🔫 Munição: {municao}") print(f"💀 Eliminações: {kills}") print(f"👾 Inimigos restantes: {inimigos}") print("\n1 - Atacar inimigo") print("2 - Procurar munição") print("3 - Usar kit médico") print("4 - Trocar arma") print("5 - Fugir") escolha = input("\nEscolha: ") if escolha == "1": if municao > 0: dano = random.randint(15, 35) municao -= 1 print(f"\n🔥 Você atacou e causou {dano} de dano!") if random.randint(1, 100) <= 70: inimigos -= 1 kills += 1 print("💀 INIMIGO ELIMINADO!") else: print("❌ O inimigo sobreviveu!") vida -= random.randint(5, 15) else: print("❌ Sem munição!") elif escolha == "2": quantidade = random.randint(5, 15) municao += quantidade print(f"🔫 Você encontrou {quantidade} munições!") elif escolha == "3": cura = random.randint(15, 30) vida = min(100, vida + cura) print(f"❤️ Você recuperou {cura} de vida!") elif escolha == "4": arma = random.choice(armas) print(f"🔫 Nova arma: {arma}") elif escolha == "5": print("🏃 Você tentou fugir!") vida -= 5 else: print("Opção inválida!") continue time.sleep(1) print("\n" + "=" * 40) print(" 🏆 FIM DA PARTIDA") print("=" * 40) if vida > 0 and inimigos == 0: print(f"🥇 BOOYAH! {nome} venceu!") else: print("💀 Você foi eliminado!") print(f"Eliminações: {kills}") print(f"Vida final: {vida}")
