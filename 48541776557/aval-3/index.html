Closures & IIFEs

function createCounter() {
  let count = 0; // Encapsulated state
  return function() {
    return ++count;
  };
}
const counter = createCounter();
console.log(counter());
console.log(counter());

ES6 Classes & Inheritance

class Animal {
  constructor(name) { this.name = name; }
  speak() { console.log(this.name + ' makes a noise.'); }
}

class Dog extends Animal {
  speak() { console.log(this.name + ' barks!'); }
}
const d = new Dog('Rex');
d.speak();

Promises & Async/Await with Fetch

async function fetchData() {
  try {
    console.log('Fetching...');
    const res = await fetch('https://jsonplaceholder.typicode.com/todos/1');
    const data = await res.json();
    console.log('Data received:', data.title);
  } catch (err) {
    console.error('Error:', err);
  }
}
fetchData();

Generators & Proxies

// Generator
function* idMaker() {
  let index = 0;
  while(index < 3) yield index++;
}
const gen = idMaker();
console.log(gen.next().value); // 0

// Proxy
const handler = {
  get: (target, prop) => prop in target ? target[prop] : 42
};
const proxyObj = new Proxy({}, handler);
console.log('Proxy default value:', proxyObj.missingKey);
