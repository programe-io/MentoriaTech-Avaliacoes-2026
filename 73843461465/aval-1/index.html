// DESAFIO GERAL - Sistema de Estoque de Produtos let estoque = []; let opcao = ""; while (opcao != "5") { opcao = prompt( "=== SISTEMA DE ESTOQUE ===\n" + "1 - Cadastrar um novo produto\n" + "2 - Listar os produtos cadastrados\n" + "3 - Alterar o valor de um produto\n" + "4 - Alterar a quantidade de um produto\n" + "5 - Sair\n" + "Escolha uma opção:" ); // 1 - CADASTRAR if (opcao == "1") { let codigo = prompt("Digite o código do produto:"); let descricao = prompt("Digite a descrição do produto:"); let quantidade = Number(prompt("Digite a quantidade:")); let valor = Number(prompt("Digite o valor:")); let produto = { codigo: codigo, descricao: descricao, quantidade: quantidade, valor: valor }; estoque.push(produto); alert("Produto cadastrado com sucesso!"); // 2 - LISTAR } else if (opcao == "2") { if (estoque.length == 0) { alert("Nenhum produto cadastrado."); } else { let lista = "=== PRODUTOS CADASTRADOS ===\n"; for (let i = 0; i < estoque.length; i++) { lista += "Código: " + estoque[i].codigo + "\n" + "Descrição: " + estoque[i].descricao + "\n" + "Quantidade: " + estoque[i].quantidade + "\n" + "Valor: R$ " + estoque[i].valor.toFixed(2) + "\n" + "-------------------------\n"; } alert(lista); } // 3 - ALTERAR VALOR } else if (opcao == "3") { let codigoBusca = prompt("Digite o código do produto que quer alterar o valor:"); let achou = false; for (let i = 0; i < estoque.length; i++) { if (estoque[i].codigo == codigoBusca) { let novoValor = Number(prompt("Digite o novo valor para " + estoque[i].descricao + ":")); estoque[i].valor = novoValor; alert("Valor alterado com sucesso!"); achou = true; } } if (achou == false) { alert("Produto não encontrado!"); } // 4 - ALTERAR QUANTIDADE } else if (opcao == "4") { let codigoBusca = prompt("Digite o código do produto que quer alterar a quantidade:"); let achou = false; for (let i = 0; i < estoque.length; i++) { if (estoque[i].codigo == codigoBusca) { let novaQtd = Number(prompt("Digite a nova quantidade para " + estoque[i].descricao + ":")); estoque[i].quantidade = novaQtd; alert("Quantidade alterada com sucesso!"); achou = true; } } if (achou == false) { alert("Produto não encontrado!"); } // 5 - SAIR } else if (opcao == "5") { alert("Saindo do sistema... Até logo!"); } else { alert("Opção inválida! Tente novamente."); } }
