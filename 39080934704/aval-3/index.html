Chave do localStorage const chavePessoas = "pessoas"; // Referências do DOM const form = document.getElementById("form-pessoa"); const inputNome = document.getElementById("nome"); const inputIdade = document.getElementById("idade"); const inputPeso = document.getElementById("peso"); const inputAltura = document.getElementById("altura"); const paragrafoErro = document.getElementById("erro"); const tbody = document.getElementById("tabela-pessoas"); // Lista inicial - carregada do localStorage let pessoas = carregarPessoas(); // Funções das aulas anteriores function formatar(valor, casas) { return valor.toFixed(casas); } function calcularIMC(peso, altura) { return peso / (altura * altura); } // 6. Classificar IMC - retorna objeto com texto e classe function classificarIMC(imc) { if (imc < 18.5) return { texto: "Abaixo do peso", classe: "abaixo" }; if (imc < 25) return { texto: "Peso normal", classe: "normal" }; if (imc < 30) return { texto: "Sobrepeso", classe: "sobrepeso" }; return { texto: "Obesidade", classe: "obesidade" }; } function criarCelula(conteudo) { const td = document.createElement("td"); td.textContent = conteudo; return td; } // 2, 3 e 4. Criar linha completa function criarLinha(pessoa) { const tr = document.createElement("tr"); // 2. IMC e classificação const imc = calcularIMC(pessoa.peso, pessoa.altura); const classificacao = classificarIMC(imc); tr.appendChild(criarCelula(pessoa.codigo)); tr.appendChild(criarCelula(pessoa.nome)); tr.appendChild(criarCelula(pessoa.idade)); // 3. Formatação de peso e altura tr.appendChild(criarCelula(formatar(pessoa.peso, 2) + " KG")); tr.appendChild(criarCelula(formatar(pessoa.altura, 2) + " m")); tr.appendChild(criarCelula(formatar(imc, 2))); // 4. Coluna de classificação com selo const tdClassificacao = document.createElement("td"); const selo = document.createElement("span"); selo.classList.add("selo", classificacao.classe); selo.textContent = classificacao.texto; tdClassificacao.appendChild(selo); tr.appendChild(tdClassificacao); return tr; } // Renderizar na tela function render() { tbody.innerHTML = ""; for (const pessoa of pessoas) { tbody.appendChild(criarLinha(pessoa)); } } // 6.
