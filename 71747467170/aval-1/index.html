function adicionarProduto(nome, quantidade, preco) { if (!nome || nome.trim().length < 5) { throw new Error("O nome do produto precisa ter pelo menos 5 caracteres."); } if (quantidade <= 0) { throw new Error("A quantidade deve ser maior que zero."); } if (preco <= 0) { throw new Error("O preço deve ser maior que zero."); } console.log("Produto cadastrado com sucesso!"); console.log("Produto:", nome); console.log("Quantidade:", quantidade); console.log("Preço: R$", preco); }
