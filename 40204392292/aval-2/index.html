/* * SISTEMA DE GERENCIAMENTO DE TAREFAS (JavaScript - Programiz) * * Cada tarefa possui: codigo, titulo, prioridade e concluida (status). * * Regras de validacao: * - titulo: no minimo 5 caracteres * - prioridade: valor entre 1 (alta) e 3 (baixa) * * Usa apenas console.log, entao roda no compilador de JavaScript do Programiz. */ // ===================== CONFIGURACOES ===================== const TITULO_MINIMO = 5; const PRIORIDADE_ALTA = 1; const PRIORIDADE_BAIXA = 3; const NOMES_PRIORIDADE = { 1: "Alta", 2: "Media", 3: "Baixa" }; // ===================== DADOS ===================== const tarefas = []; let proximoCodigo = 1; // ===================== VALIDACOES ===================== function tituloValido(titulo) { return typeof titulo === "string" && titulo.trim().length >= TITULO_MINIMO; } function prioridadeValida(prioridade) { return ( Number.isInteger(prioridade) && prioridade >= PRIORIDADE_ALTA && prioridade <= PRIORIDADE_BAIXA ); } function buscarTarefa(codigo) { return tarefas.find(function (tarefa) { return tarefa.codigo === codigo; }); } // ===================== FUNCIONALIDADES ===================== // 1. Cadastrar uma nova tarefa function cadastrarTarefa(titulo, prioridade) { if (!tituloValido(titulo)) { console.log("[ERRO] O titulo deve ter no minimo " + TITULO_MINIMO + " caracteres."); return null; } if (!prioridadeValida(prioridade)) { console.log("[ERRO] A prioridade deve ser um valor entre 1 (alta) e 3 (baixa)."); return null; } const tarefa = { codigo: proximoCodigo, titulo: titulo.trim(), prioridade: prioridade, concluida: false, }; proximoCodigo++; tarefas.push(tarefa); console.log('[OK] Tarefa "' + tarefa.titulo + '" cadastrada com o codigo ' + tarefa.codigo + "."); return tarefa; } // 2. Listar as tarefas cadastradas function listarTarefas() { console.log("\n================ TAREFAS CADASTRADAS ================"); if (tarefas.length === 0) { console.log("Nenhuma tarefa cadastrada."); } else { tarefas.forEach(function (t) { const status = t.concluida ? "Concluida" : "Pendente"; console.log( "Codigo: " + t.codigo + " | Titulo: " + t.titulo + " | Prioridade: " + t.prioridade + " (" + NOMES_PRIORIDADE[t.prioridade] + ")" + " | Status: " + status ); }); } console.log("=====================================================\n"); } // 3. Marcar uma tarefa como concluida function concluirTarefa(codigo) { const tarefa = buscarTarefa(codigo); if (!tarefa) { console.log("[ERRO] Tarefa de codigo " + codigo + " nao encontrada."); return false; } if (tarefa.concluida) { console.log("[AVISO] A tarefa " + codigo + " ja estava concluida."); return false; } tarefa.concluida = true; console.log("[OK] Tarefa " + codigo + " marcada como concluida."); return true; } // 4. Alterar a prioridade de uma tarefa function alterarPrioridade(codigo, novaPrioridade) { const tarefa = buscarTarefa(codigo); if (!tarefa) { console.log("[ERRO] Tarefa de codigo " + codigo + " nao encontrada."); return false; } if (!prioridadeValida(novaPrioridade)) { console.log("[ERRO] A prioridade deve ser um valor entre 1 (alta) e 3 (baixa)."); return false; } tarefa.prioridade = novaPrioridade; console.log( "[OK] Prioridade da tarefa " + codigo + " alterada para " + novaPrioridade + " (" + NOMES_PRIORIDADE[novaPrioridade] + ")." ); return true; } // ===================== DEMONSTRACAO ===================== console.log("--- Cadastrando tarefas ---"); cadastrarTarefa("Estudar JavaScript", 1); cadastrarTarefa("Lavar o carro", 3); cadastrarTarefa("Pagar contas", 2); console.log("\n--- Testando validacoes ---"); cadastrarTarefa("Abc", 2); // titulo com menos de 5 caracteres cadastrarTarefa("Fazer compras", 5); // prioridade fora do intervalo 1 a 3 listarTarefas(); console.log("--- Concluindo e alterando ---"); concluirTarefa(1); concluirTarefa(1); // ja concluida alterarPrioridade(2, 1); alterarPrioridade(2, 7); // prioridade invalida concluirTarefa(99); // codigo inexistente listarTarefas(); // ===================== CONFIRMACAO (TESTES) ===================== function verificar(descricao, condicao) { console.log("[" + (condicao ? "OK" : "FALHOU") + "] " + descricao); } console.log("=============== CONFIRMACAO DOS RESULTADOS ==============="); verificar("3 tarefas foram cadastradas (as invalidas foram recusadas)", tarefas.length === 3); verificar("Titulo com menos de 5 caracteres e recusado", !tituloValido("Abc")); verificar("Titulo com 5 caracteres e aceito", tituloValido("Abcde")); verificar("Prioridade 0 e recusada", !prioridadeValida(0)); verificar("Prioridade 4 e recusada", !prioridadeValida(4)); verificar("Prioridades 1, 2 e 3 sao aceitas", [1, 2, 3].every(prioridadeValida)); verificar("Tarefa 1 esta concluida", buscarTarefa(1).concluida === true); verificar("Tarefa 2 continua pendente", buscarTarefa(2).concluida === false); verificar("Prioridade da tarefa 2 mudou de 3 para 1", buscarTarefa(2).prioridade === 1); verificar("Prioridade da tarefa 3 continua 2", buscarTarefa(3).prioridade === 2); verificar("Tarefa 99 nao existe", buscarTarefa(99) === undefined);
