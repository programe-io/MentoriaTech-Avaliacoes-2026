let produtos = [];

// Função assíncrona para gerar o código automaticamente
async function gerarCodigo() {
    await new Promise(resolve => setTimeout(resolve, 100));
        return produtos.length + 1;
        }

        // Cadastrar produto
        async function cadastrarProduto() {
            let codigo = await gerarCodigo();

                let descricao = prompt("Digite a descrição do produto:");

                    if (!descricao || descricao.length < 5) {
                            alert("A descrição deve ter no mínimo 5 caracteres!");
                                    return;
                                        }

                                            let quantidade = Number(prompt("Digite a quantidade:"));
                                                let valor = Number(prompt("Digite o valor:"));

                                                    if (quantidade < 0 || valor < 0 || isNaN(quantidade) || isNaN(valor)) {
                                                            alert("Quantidade ou valor inválido!");
                                                                    return;
                                                                        }

                                                                            let produto = {
                                                                                    codigo: codigo,
                                                                                            descricao: descricao,
                                                                                                    quantidade: quantidade,
                                                                                                            valor: valor
                                                                                                                };

                                                                                                                    produtos.push(produto);

                                                                                                                        alert("Produto cadastrado com sucesso! Código: " + codigo);
                                                                                                                        }

                                                                                                                        // Listar produtos
                                                                                                                        function listarProdutos() {
                                                                                                                            if (produtos.length === 0) {
                                                                                                                                    alert("Nenhum produto cadastrado!");
                                                                                                                                            return;
                                                                                                                                                }

                                                                                                                                                    let lista = "";

                                                                                                                                                        produtos.forEach(function(produto) {
                                                                                                                                                                lista +=
                                                                                                                                                                            "Código: " + produto.codigo +
                                                                                                                                                                                        "\nDescrição: " + produto.descricao +
                                                                                                                                                                                                    "\nQuantidade: " + produto.quantidade +
                                                                                                                                                                                                                "\nValor: R$ " + produto.valor.toFixed(2) +
                                                                                                                                                                                                                            "\n-------------------\n";
                                                                                                                                                                                                                                });

                                                                                                                                                                                                                                    alert(lista);
                                                                                                                                                                                                                                    }

                                                                                                                                                                                                                                    // Alterar valor do produto
                                                                                                                                                                                                                                    function alterarValor() {
                                                                                                                                                                                                                                        let codigo = Number(prompt("Digite o código do produto:"));

                                                                                                                                                                                                                                            let produto = produtos.find(p => p.codigo === codigo);

                                                                                                                                                                                                                                                if (produto) {
                                                                                                                                                                                                                                                        let novoValor = Number(prompt("Digite o novo valor:"));

                                                                                                                                                                                                                                                                if (novoValor < 0 || isNaN(novoValor)) {
                                                                                                                                                                                                                                                                            alert("Valor inválido!");
                                                                                                                                                                                                                                                                                        return;
                                                                                                                                                                                                                                                                                                }

                                                                                                                                                                                                                                                                                                        produto.valor = novoValor;
                                                                                                                                                                                                                                                                                                                alert("Valor alterado com sucesso!");
                                                                                                                                                                                                                                                                                                                    } else {
                                                                                                                                                                                                                                                                                                                            alert("Produto não encontrado!");
                                                                                                                                                                                                                                                                                                                                }
                                                                                                                                                                                                                                                                                                                                }

                                                                                                                                                                                                                                                                                                                                // Alterar quantidade do produto
                                                                                                                                                                                                                                                                                                                                function alterarQuantidade() {
                                                                                                                                                                                                                                                                                                                                    let codigo = Number(prompt("Digite o código do produto:"));

                                                                                                                                                                                                                                                                                                                                        let produto = produtos.find(p => p.codigo === codigo);

                                                                                                                                                                                                                                                                                                                                            if (produto) {
                                                                                                                                                                                                                                                                                                                                                    let novaQuantidade = Number(prompt("Digite a nova quantidade:"));

                                                                                                                                                                                                                                                                                                                                                            if (novaQuantidade < 0 || isNaN(novaQuantidade)) {
                                                                                                                                                                                                                                                                                                                                                                        alert("Quantidade inválida!");
                                                                                                                                                                                                                                                                                                                                                                                    return;
                                                                                                                                                                                                                                                                                                                                                                                            }

                                                                                                                                                                                                                                                                                                                                                                                                    produto.quantidade = novaQuantidade;
                                                                                                                                                                                                                                                                                                                                                                                                            alert("Quantidade alterada com sucesso!");
                                                                                                                                                                                                                                                                                                                                                                                                                } else {
                                                                                                                                                                                                                                                                                                                                                                                                                        alert("Produto não encontrado!");
                                                                                                                                                                                                                                                                                                                                                                                                                            }
                                                                                                                                                                                                                                                                                                                                                                                                                            }

                                                                                                                                                                                                                                                                                                                                                                                                                                let opcao;

                                                                                                                                                                                                                                                                                                                                                                                                                                                        "SISTEMA DE ESTOQUE\n\n" +

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            }

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                } while (opcao !== "5");
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                }

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                // Executar o sistema
