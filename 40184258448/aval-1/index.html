// ============================================================ // LISTA DE PRODUTOS - VERSÃO PARA O PROGRAMIZ (JavaScript) // Cole em: https://www.programiz.com/javascript/online-compiler/ // Não usa prompt(): tudo roda só com console.log. // Para testar outros casos, altere as chamadas na seção "USO" no final. // // Regras: // - nome com no mínimo 3 caracteres // - preço maior que 0 // - quantidade inteira maior ou igual a 0 // ============================================================ var produtos = []; var proximoId = 1; // ---------- Validações ---------- function nomeValido(nome) { return typeof nome === "string" && nome.trim().length >= 3; } function precoValido(preco) { return typeof preco === "number" && preco > 0; } function quantidadeValida(qtd) { return Number.isInteger(qtd) && qtd >= 0; } function formatarPreco(valor) { return "R$ " + valor.toFixed(2).replace(".", ","); } function buscarPorId(id) { for (var i = 0; i < produtos.length; i++) { if (produtos[i].id === id) return produtos[i]; } console.log("Erro: produto #" + id + " não encontrado."); return null; } // ---------- 1) Cadastrar produto ---------- function cadastrarProduto(nome, preco, quantidade) { if (!nomeValido(nome)) { console.log('Erro: nome "' + nome + '" inválido (mínimo 3 caracteres).'); return; } if (!precoValido(preco)) { console.log("Erro: preço inválido (deve ser maior que 0)."); return; } if (!quantidadeValida(quantidade)) { console.log("Erro: quantidade inválida (use inteiro maior ou igual a 0)."); return; } var produto = { id: proximoId, nome: nome.trim(), preco: preco, quantidade: quantidade }; proximoId++; produtos.push(produto); console.log("OK: produto #" + produto.id + ' "' + produto.nome + '" cadastrado.'); } // ---------- 2) Listar produtos ---------- function listarProdutos() { console.log(""); console.log("===== LISTA DE PRODUTOS ====="); if (produtos.length === 0) { console.log("Nenhum produto cadastrado."); return; } for (var i = 0; i < produtos.length; i++) { var p = produtos[i]; var aviso = p.quantidade === 0 ? " (SEM ESTOQUE)" : ""; console.log( "#" + p.id + " - " + p.nome + " | " + formatarPreco(p.preco) + " | Qtd: " + p.quantidade + aviso ); } console.log("============================="); } // ---------- 3) Alterar preço ---------- function alterarPreco(id, novoPreco) { var p = buscarPorId(id); if (!p) return; if (!precoValido(novoPreco)) { console.log("Erro: preço inválido (deve ser maior que 0)."); return; } p.preco = novoPreco; console.log("OK: preço de " + p.nome + " alterado para " + formatarPreco(novoPreco) + "."); } // ---------- 4) Entrada de estoque ---------- function adicionarEstoque(id, quantidade) { var p = buscarPorId(id); if (!p) return; if (!Number.isInteger(quantidade) || quantidade <= 0) { console.log("Erro: informe uma quantidade inteira maior que 0."); return; } p.quantidade += quantidade; console.log("OK: +" + quantidade + " de " + p.nome + ". Total: " + p.quantidade + "."); } // ---------- 5) Vender produto ---------- function venderProduto(id, quantidade) { var p = buscarPorId(id); if (!p) return; if (!Number.isInteger(quantidade) || quantidade <= 0) { console.log("Erro: informe uma quantidade inteira maior que 0."); return; } if (quantidade > p.quantidade) { console.log("Erro: estoque insuficiente de " + p.nome + " (disponível: " + p.quantidade + ")."); return; } p.quantidade -= quantidade; console.log("OK: venda de " + quantidade + "x " + p.nome + " = " + formatarPreco(quantidade * p.preco) + "."); } // ---------- 6) Remover produto ---------- function removerProduto(id) { for (var i = 0; i < produtos.length; i++) { if (produtos[i].id === id) { var nome = produtos[i].nome; produtos.splice(i, 1); console.log("OK: produto " + nome + " removido."); return; } } console.log("Erro: produto #" + id + " não encontrado."); } // ---------- 7) Relatório ---------- function relatorio() { var total = 0; var semEstoque = []; for (var i = 0; i < produtos.length; i++) { total += produtos[i].preco * produtos[i].quantidade; if (produtos[i].quantidade === 0) semEstoque.push(produtos[i].nome); } console.log(""); console.log("===== RELATÓRIO ====="); console.log("Produtos cadastrados: " + produtos.length); console.log("Valor total em estoque: " + formatarPreco(total)); console.log("Sem estoque: " + (semEstoque.length > 0 ? semEstoque.join(", ") : "nenhum")); console.log("====================="); } // ============================================================ // USO (altere à vontade para testar) // ============================================================ console.log("*** LISTA DE PRODUTOS ***"); // Cadastrar cadastrarProduto("Arroz", 25.9, 10); cadastrarProduto("Feijão", 8.5, 4); cadastrarProduto("Café", 18, 0); cadastrarProduto("Ab", 5, 1); // erro: nome com menos de 3 caracteres cadastrarProduto("Sal", -2, 1); // erro: preço inválido cadastrarProduto("Açúcar", 4.75, 2.5); // erro: quantidade não inteira // Listar listarProdutos(); // Vender e repor estoque venderProduto(1, 3); venderProduto(2, 10); // erro: estoque insuficiente adicionarEstoque(3, 8); // Alterar preço e remover alterarPreco(2, 9.2); removerProduto(99); // erro: não existe // Resultado final listarProdutos(); relatorio();
