import pandas as pd # Criando um conjunto de dados para uma nova visualização dados_mercado_v2 = { 'Código': ['010', '011', '012', '013', '014', '015'], 'Produto': ['Arroz (5kg)', 'Feijão (1kg)', 'Óleo de Soja', 'Açúcar (1kg)', 'Café (500g)', 'Leite (1L)'], 'Preço (R$)': [29.90, 8.50, 7.20, 4.30, 16.90, 5.49] } df_v2 = pd.DataFrame(dados_mercado_v2) total_v2 = df_v2['Preço (R$)'].sum() df_v2['Porcentagem (%)'] = (df_v2['Preço (R$)'] / total_v2) * 100 # Estilização moderna com barras horizontais na própria tabela tabela_barras = df_v2.style.format({ 'Preço (R$)': 'R$ {:.2f}', 'Porcentagem (%)': '{:.2f}%' }).bar(subset=['Porcentagem (%)'], color='#2ecc71', vmin=0, vmax=100) display(tabela_barras)
