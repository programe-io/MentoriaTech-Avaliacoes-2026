function curtir(botao) { if (botao.dataset.curtido === "sim") { botao.innerHTML = "🤍 Curtir"; botao.dataset.curtido = "nao"; } else { botao.innerHTML = "❤️ Curtido"; botao.dataset.curtido = "sim"; } } function publicar() { const usuario = document.getElementById("usuario").value.trim(); const legenda = document.getElementById("legenda").value.trim(); if (usuario === "" || legenda === "") { alert("Preencha seu nome e sua legenda!"); return; } const feed = document.getElementById("feed"); const post = document.createElement("article"); post.className = "publicacao"; const nome = document.createElement("div"); nome.className = "p-3 font-bold"; nome.textContent = "@" + usuario; const imagem = document.createElement("div"); imagem.className = "imagem"; imagem.textContent = "📸"; const conteudo = document.createElement("div"); conteudo.className = "conteudo"; const botao = document.createElement("button"); botao.innerHTML = "🤍 Curtir"; botao.onclick = function() { curtir(botao); }; const texto = document.createElement("p"); texto.className = "mt-2"; texto.textContent = "@" + usuario + " " + legenda; conteudo.appendChild(botao); conteudo.appendChild(texto); post.appendChild(nome); post.appendChild(imagem); post.appendChild(conteudo); feed.prepend(post); document.getElementById("usuario").value = ""; document.getElementById("legenda").value = ""; } }
