from IPython.display import clear_output import sys import time # Lista que armazenará os produtos produtos = [] def validar_produto(descricao, quantidade, valor): if len(descricao) < 5: raise ValueError( "A descrição deve ter, no mínimo, cinco caracteres." ) if quantidade < 1: raise ValueError( "A quantidade deve ser maior que zero." ) if valor < 0: raise ValueError( "O valor deve ser maior ou igual a zero." ) def buscar_produto(codigo): for produto in produtos: if produto["codigo"] == codigo: return produto return None def mostrar_tela(mensagem=""): clear_output(wait=False) print("========== SISTEMA DE ESTOQUE ==========") print("1 - Cadastrar produto") print("2 - Listar produtos") print("3 - Atualizar valor") print("4 - Adicionar quantidade ao estoque") print("5 - Sair") print("\n========== PRODUTOS CADASTRADOS ==========") if len(produtos) == 0: print("Nenhum produto cadastrado.") else: for produto in produtos: print("-" * 42) print(f"Código: {produto['codigo']}") print(f"Descrição: {produto['descricao']}") print(f"Quantidade: {produto['quantidade']}") print(f"Valor: R$ {produto['valor']:.2f}") print("-" * 42) if mensagem: print(f"\n{mensagem}") sys.stdout.flush() def cadastrar_produto(): print("\n--- CADASTRAR PRODUTO ---") descricao = input("Descrição do produto: ").strip() quantidade = int(input("Quantidade inicial: ")) valor = float( input("Valor do produto: R$ ").replace(",", ".") ) validar_produto(descricao, quantidade, valor) novo_produto = { "codigo": len(produtos) + 1, "descricao": descricao, "quantidade": quantidade, "valor": valor } produtos.append(novo_produto) return ( "Produto cadastrado com sucesso! " f"Código gerado: {novo_produto['codigo']}." ) def atualizar_valor(): print("\n--- ATUALIZAR VALOR ---") codigo = int(input("Código do produto: ")) produto = buscar_produto(codigo) if produto is None: raise ValueError("Produto não encontrado.") novo_valor = float( input("Novo valor: R$ ").replace(",", ".") ) if novo_valor < 0: raise ValueError( "O valor deve ser maior ou igual a zero." ) produto["valor"] = novo_valor return "Valor atualizado com sucesso!" def adicionar_quantidade(): print("\n--- ADICIONAR QUANTIDADE ---") codigo = int(input("Código do produto: ")) produto = buscar_produto(codigo) if produto is None: raise ValueError("Produto não encontrado.") quantidade_adicionada = int( input("Quantidade que será adicionada: ") ) if quantidade_adicionada < 1: raise ValueError( "A quantidade deve ser maior que zero." ) produto["quantidade"] += quantidade_adicionada return ( "Quantidade atualizada com sucesso! " f"Estoque atual: {produto['quantidade']} unidades." ) def iniciar_sistema(): mensagem = "" while True: mostrar_tela(mensagem) time.sleep(0.1) print() sys.stdout.flush() opcao = input("Escolha uma opção: ").strip() try: if opcao == "1": mensagem = cadastrar_produto() continue if opcao == "2": mensagem = "Lista de produtos atualizada." continue if opcao == "3": mensagem = atualizar_valor() continue if opcao == "4": mensagem = adicionar_quantidade() continue if opcao == "5": mostrar_tela("Sistema encerrado.") break mensagem = "Opção inválida. Digite um número de 1 a 5." except ValueError as erro: mensagem = f"Erro: {erro}" except Exception as erro: mensagem = f"Erro inesperado: {erro}" # Inicia o programa iniciar_sistema()o
