body { font-family: Arial; background: #eaf0f5; } .container { background: white; max-width: 500px; margin: 30px auto; padding: 20px; border-radius: 10px; } h1 { text-align: center; color: blue; } input { width: 95%; padding: 10px; margin: 5px; } button { width: 100%; padding: 10px; margin: 5px; background: blue; color: white; border: none; cursor: pointer; } .produto { background: #ddd; padding: 10px; margin: 10px; } let produtos = []; function cadastrar() { let codigo = document.getElementById("codigo").value; let descricao = document.getElementById("descricao").value; let quantidade = Number(document.getElementById("quantidade").value); let valor = Number(document.getElementById("valor").value); if (!codigo || !descricao) { alert("Preencha os campos!"); return; } produtos.push({codigo, descricao, quantidade, valor}); alert("Produto cadastrado!"); listar(); } function listar() { let lista = document.getElementById("lista"); lista.innerHTML = ""; produtos.forEach(p => { lista.innerHTML += `
Código: ${p.codigo}
Descrição: ${p.descricao}
Quantidade: ${p.quantidade}
Valor: R$ ${p.valor.toFixed(2)}
`; }); } function alterarValor() { let codigo = document.getElementById("codigoAlterar").value; let produto = produtos.find(p => p.codigo == codigo); if (produto) { produto.valor = Number(document.getElementById("novoValor").value); listar(); alert("Valor alterado!"); } else { alert("Produto não encontrado!"); } } function alterarQuantidade() { let codigo = document.getElementById("codigoAlterar").value; let produto = produtos.find(p => p.codigo == codigo); if (produto) { produto.quantidade = Number(document.getElementById("novaQuantidade").value); listar(); alert("Quantidade alterada!"); } else { alert("Produto não encontrado!"); } }
