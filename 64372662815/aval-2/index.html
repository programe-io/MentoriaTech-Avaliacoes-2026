const readline = require("readline"); const rl = readline.createInterface({ input: process.stdin, output: process.stdout }); let tarefas = []; let proximoCodigo = 1; // Função para ler informações function perguntar(mensagem) { return new Promise(resolve => { rl.question(mensagem, resolve); }); } // Cadastrar nova tarefa async function cadastrarTarefa() { console.log("\n=== CADASTRAR TAREFA ==="); let titulo = await perguntar("Digite o título da tarefa: "); if (titulo.trim().length < 5) { console.log("Erro: o título deve ter no mínimo 5 caracteres!"); return; } console.log("\nPrioridades:"); console.log("1 - Alta"); console.log("2 - Média"); console.log("3 - Baixa"); let prioridade = Number( await perguntar("Escolha a prioridade: ") ); if (![1, 2, 3].includes(prioridade)) { console.log("Erro: prioridade inválida!"); return; } let tarefa = { codigo: proximoCodigo++, titulo: titulo.trim(), prioridade: prioridade, status: "Pendente" }; tarefas.push(tarefa); console.log("\nTarefa cadastrada com sucesso!"); } // Listar tarefas cadastradas function listarTarefas() { console.log("\n=== LISTA DE TAREFAS ==="); if (tarefas.length === 0) { console.log("Nenhuma tarefa cadastrada."); return; } tarefas.forEach(tarefa => { console.log("----------------------------"); console.log("Código:", tarefa.codigo); console.log("Título:", tarefa.titulo); let nomePrioridade = { 1: "Alta", 2: "Média", 3: "Baixa" }; console.log("Prioridade:", nomePrioridade[tarefa.prioridade]); console.log("Status:", tarefa.status); }); console.log("----------------------------"); } // Marcar tarefa como concluída async function concluirTarefa() { console.log("\n=== CONCLUIR TAREFA ==="); let codigo = Number( await perguntar("Digite o código da tarefa: ") ); let tarefa = tarefas.find(t => t.codigo === codigo); if (!tarefa) { console.log("Tarefa não encontrada!"); return; } tarefa.status = "Concluída"; console.log("Tarefa concluída com sucesso!"); } // Alterar prioridade da tarefa async function alterarPrioridade() { console.log("\n=== ALTERAR PRIORIDADE ==="); let codigo = Number( await perguntar("Digite o código da tarefa: ") ); let tarefa = tarefas.find(t => t.codigo === codigo); if (!tarefa) { console.log("Tarefa não encontrada!"); return; } console.log("\n1 - Alta"); console.log("2 - Média"); console.log("3 - Baixa"); let novaPrioridade = Number( await perguntar("Digite a nova prioridade: ") ); if (![1, 2, 3].includes(novaPrioridade)) { console.log("Prioridade inválida!"); return; } tarefa.prioridade = novaPrioridade; console.log("Prioridade alterada com sucesso!"); } // Menu principal async function menu() { let opcao; do { console.log("\n================================"); console.log(" SISTEMA DE GERENCIAMENTO"); console.log("================================"); console.log("1 - Cadastrar nova tarefa"); console.log("2 - Listar tarefas"); console.log("3 - Concluir tarefa"); console.log("4 - Alterar prioridade"); console.log("0 - Sair"); console.log("================================"); opcao = await perguntar("Escolha uma opção: "); switch (opcao) { case "1": await cadastrarTarefa(); break; case "2": listarTarefas(); break; case "3": await concluirTarefa(); break; case "4": await alterarPrioridade(); break; case "0": console.log("Sistema encerrado. Até mais!"); break; default: console.log("Opção inválida!"); } } while (opcao !== "0"); rl.close(); } // Executar o sistema menu();
