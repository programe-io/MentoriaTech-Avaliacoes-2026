let clientes = []; let agendamentos = []; // 1. Cadastrar cliente function cadastrarCliente(nome, telefone) { if (!nome || !telefone) { console.log("Preencha todos os dados!"); return; } clientes.push({ codigo: clientes.length + 1, nome: nome, telefone: telefone }); console.log("Cliente cadastrado com sucesso!"); } // 2. Listar clientes function listarClientes() { console.table(clientes); } // 3. Criar agendamento function criarAgendamento(codigoCliente, servico, data) { let cliente = clientes.find(c => c.codigo === codigoCliente); if (!cliente || !servico || !data) { console.log("Dados do agendamento inválidos!"); return; } agendamentos.push({ codigo: agendamentos.length + 1, cliente: cliente.nome, servico: servico, data: data }); console.log("Agendamento realizado!"); } // 4. Listar agendamentos function listarAgendamentos() { console.table(agendamentos); } // 5. Cancelar agendamento function cancelarAgendamento(codigo) { let indice = agendamentos.findIndex(a => a.codigo === codigo); if (indice === -1) { console.log("Agendamento não encontrado!"); return; } agendamentos.splice(indice, 1); console.log("Agendamento cancelado!"); }
