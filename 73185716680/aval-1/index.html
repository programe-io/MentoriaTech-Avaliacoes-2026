let produtos = []; let proximoCodigo = 1; // 1. Validar os dados do produto function validarProduto(descricao, quantidade, valor) { if (!descricao || descricao.trim() === "") { return false; } if (!Number.isInteger(quantidade) || quantidade < 0) { return false; } if (typeof valor !== "number" || valor < 0 || !Number.isFinite(valor)) { return false; } return true; } // 2. Cadastrar um novo produto function cadastrarProduto(descricao, quantidade, valor) { if (!validarProduto(descricao, quantidade, valor)) { console.log("Dados inválidos!"); return; } let produto = { codigo: proximoCodigo++, descricao: descricao.trim(), quantidade: quantidade, valor: valor }; produtos.push(produto); console.log("Produto cadastrado com sucesso!"); } // 3. Listar os produtos cadastrados function listarProdutos() { if (produtos.length === 0) { console.log("Nenhum produto cadastrado."); return; } console.table(produtos); } // 4. Atualizar o valor de um produto function atualizarValor(codigoProduto, novoValor) { let produto = produtos.find(p => p.codigo === codigoProduto); if (!produto || typeof novoValor !== "number" || !Number.isFinite(novoValor) || novoValor < 0) { console.log("Produto ou valor inválido!"); return; } produto.valor = novoValor; console.log("Valor atualizado com sucesso!"); } // 5. Atualizar a quantidade de um produto function atualizarQuantidade(codigoProduto, novaQuantidade) { let produto = produtos.find(p => p.codigo === codigoProduto); if (!produto || !Number.isInteger(novaQuantidade) || novaQuantidade < 0) { console.log("Produto ou quantidade inválida!"); return; } produto.quantidade = novaQuantidade; console.log("Quantidade atualizada com sucesso!"); } // Exemplos de utilização cadastrarProduto("Caderno", 10, 15.50); cadastrarProduto("Caneta", 20, 2.50); listarProdutos(); atualizarValor(1, 18.00); atualizarQuantidade(2, 30); listarProdutos();
