import cv2 import numpy as np def endireitar_e_limpar_imagem(caminho_imagem): # 1. Carrega a imagem original (que estava de ponta-cabeça e borrada) img = cv2.imread(caminho_imagem) if img is None: print("Erro: Não foi possível carregar a imagem.") return # 2. Corrige a orientação (Rotaciona 180 graus para desinverter) # Isso simula a transição da Imagem 1 para a Imagem 3 img_correta = cv2.rotate(img, cv2.ROTATE_180) # 3. Transforma em escala de cinza para facilitar a leitura do texto cinza = cv2.cvtColor(img_correta, cv2.COLOR_BGR2GRAY) # 4. Remove ruídos visuais usando um filtro Gaussiano # Isso limpa as distorções presentes nas etapas intermediárias borrado = cv2.GaussianBlur(cinza, (3, 3), 0) # 5. Aplica Limiarização (Thresholding) para destacar o texto # Transforma o fundo em branco puro e as letras em preto nítido (Etapa Final) _, imagem_final = cv2.threshold(borrado, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU) # Salva o resultado final perfeitamente legível cv2.imwrite('resultado_desembaralhado.png', imagem_final) print("Imagem processada e salva com sucesso como 'resultado_desembaralhado.png'!") # Para rodar o código, basta substituir pelo nome do seu arquivo de imagem # endireitar_e_limpar_imagem('sua_imagem_invertida.png')
