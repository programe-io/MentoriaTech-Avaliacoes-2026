// Define o estado emocional inicial do usuário let userState = { mood: "sad", energyLevel: 40, confidence: 50, isAwesome: false }; // Função para tratar o estado de tristeza function stopBeingSad() { console.log("Iniciando protocolo: stopBeingSad()..."); // Altera os níveis do estado do usuário userState.mood = "neutral"; userState.energyLevel += 30; userState.confidence += 20; console.log("Resultado: A tristeza foi interrompida com sucesso."); console.log(`Status atual: Humor: ${userState.mood} | Confiança: ${userState.confidence}`); } // Função para ativar o modo incrível function beAwesome() { console.log("Atenção: Modo extraordinário detectado!"); console.log("Iniciando protocolo: beAwesome()..."); // Eleva os atributos ao nível máximo userState.mood = "happy"; userState.energyLevel = 100; userState.confidence = 100; userState.isAwesome = true; console.log("Resultado: Você atingiu o nível máximo de grandiosidade!"); console.log("🚀 Lembre-se: Você é incrível todos os dias!"); } // Execução da estrutura condicional baseada nos inputs visuais if (userState.mood === "sad") { stopBeingSad(); // Uma segunda verificação para garantir a transição de humor if (userState.mood !== "sad") { console.log("Transição concluída. Agora você está pronto para o próximo nível."); beAwesome(); } } else { beAwesome(); }
