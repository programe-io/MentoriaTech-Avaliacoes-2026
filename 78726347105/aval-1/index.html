# Dicionário para armazenar o estoque: { 'codigo': {'nome': '...', 'qtd': 0, 'preco': 0.0} } estoque = {} def cadastrar_produto(): codigo = input("Digite o código do produto: ") if codigo in estoque: print("Produto já cadastrado!") return nome = input("Nome do produto: ") qtd = int(input("Quantidade inicial: ")) preco = float(input("Preço unitário (R$): ")) estoque[codigo] = {"nome": nome, "qtd": qtd, "preco": preco} print("Produto cadastrado com sucesso!") def registrar_movimentacao(tipo): codigo = input("Digite o código do produto: ") if codigo not in estoque: print("Produto não encontrado!") return qtd = int(input(f"Quantidade para {'entrada' if tipo == 'e' else 'saída'}: ")) if tipo == 'e': estoque[codigo]["qtd"] += qtd print("Entrada registrada!") elif tipo == 's': if qtd > estoque[codigo]["qtd"]: print("Estoque insuficiente!") else: estoque[codigo]["qtd"] -= qtd print("Saída registrada!") def listar_estoque(): print("\n--- ESTOQUE ATUAL ---") if not estoque: print("Nenhum produto cadastrado.") for codigo, dados in estoque.items(): print(f"Cód: {codigo} | Produto: {dados['nome']} | Qtd: {dados['qtd']} | Preço: R$ {dados['preco']:.2f}") print("-" * 25) def main(): while True: print("\n1. Cadastrar produto") print("2. Registrar entrada") print("3. Registrar saída") print("4. Listar estoque") print("5. Sair") opcao = input("Escolha uma opção: ") if opcao == "1": cadastrar_produto() elif opcao == "2": registrar_movimentacao('e') elif opcao == "3": registrar_movimentacao('s') elif opcao == "4": listar_estoque() elif opcao == "5": print("Saindo...") break else: print("Opção inválida!") if __name__ == "__main__": main()
