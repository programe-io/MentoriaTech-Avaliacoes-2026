function cadastrarTarefa(codigo, titulo, prioridade) { if (titulo.length < 5) { console.log("Erro: o título deve ter no mínimo 5 caracteres."); return; } if (prioridade < 1 || prioridade > 3) { console.log("Erro: prioridade deve ser entre 1 e 3."); return; } tarefas.push({ codigo: codigo, titulo: titulo, prioridade: prioridade, concluida: false }); console.log("Tarefa cadastrada com sucesso!"); } function listarTarefas() { console.log(tarefas); } function concluirTarefa(codigo) { let tarefa = tarefas.find(t => t.codigo === codigo); if (tarefa) { tarefa.concluida = true; console.log("Tarefa concluída!"); } else { console.log("Tarefa não encontrada."); } } function alterarPrioridade(codigo, novaPrioridade) { let tarefa = tarefas.find(t => t.codigo === codigo); if (tarefa && novaPrioridade >= 1 && novaPrioridade <= 3) { tarefa.prioridade = novaPrioridade; console.log("Prioridade alterada!"); } else { console.log("Dados inválidos."); } } // Testando o sistema cadastrarTarefa(1, "Estudar JavaScript", 1); cadastrarTarefa(2, "Fazer atividade", 2); listarTarefas(); concluirTarefa(1); alterarPrioridade(2, 3); listarTarefas();
