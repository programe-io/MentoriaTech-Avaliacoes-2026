let tarefas = []; let codigoAtual = 0; // 1. Validar dados da tarefa function validarDadosDaTarefa(titulo, prioridade) { if (titulo.length < 5) { throw new Error("O título deve ter pelo menos 5 caracteres"); } if (prioridade < 1 || prioridade > 3) { throw new Error("A prioridade deve ser entre 1 e 3"); } } // 2. Cadastrar tarefa function cadastrarTarefa() { let titulo = document.getElementById("titulo").value; let prioridade = Number(document.getElementById("prioridade").value); try { validarDadosDaTarefa(titulo, prioridade); codigoAtual++; let novaTarefa = { codigo: codigoAtual, titulo: titulo, prioridade: prioridade, status: true }; tarefas.push(novaTarefa); listarTarefas(); document.getElementById("titulo").value = ""; document.getElementById("prioridade").value = ""; } catch (erro) { alert(erro.message); } } // 3. Listar tarefas function listarTarefas() { return tarefas; } // 4. Buscar tarefa pelo código function buscarTarefa(codigo) { const tarefa = tarefas.find( (item) => item.codigo === codigo ); if (!tarefa) { throw new Error("Tarefa não encontrada"); } return tarefa; } // 5. Concluir tarefa function concluirTarefa(codigo) { try { const tarefa = buscarTarefa(codigo); if (tarefa.status === false) { throw new Error("Essa tarefa já foi concluída"); } tarefa.status = false; listarNaTela(); } catch (erro) { alert(erro.message); } } // 6. Alterar prioridade function alterarPrioridade(codigo) { try { const tarefa = buscarTarefa(codigo); let novaPrioridade = Number( prompt("Digite a nova prioridade (1, 2 ou 3):") ); validarDadosDaTarefa( tarefa.titulo, novaPrioridade ); tarefa.prioridade = novaPrioridade; listarNaTela(); } catch (erro) { alert(erro.message); } } // 7. Exibir tarefas na tabela function listarNaTela() { let tabela = document.getElementById("listaTarefas"); tabela.innerHTML = ""; listarTarefas().forEach(function(tarefa) { let status = tarefa.status ? "Em andamento" : "Concluída"; let classe = tarefa.status ? "pendente" : "concluida"; tabela.innerHTML += ` ${tarefa.codigo} ${tarefa.titulo} ${tarefa.prioridade} ${status} `; }); }
