let produtos = []; function cadastrarProduto() { let codigo = prompt("Digite o código do produto:"); let descricao = prompt("Digite a descrição do produto:"); let quantidade = Number(prompt("Digite a quantidade:")); let valor = Number(prompt("Digite o valor do produto:")); produtos.push({ codigo: codigo, descricao: descricao, quantidade: quantidade, valor: valor }); alert("Produto cadastrado com sucesso!"); } function listarProdutos() { let lista = ""; produtos.forEach(produto => { lista += `Código: ${produto.codigo} Descrição: ${produto.descricao} Quantidade: ${produto.quantidade} Valor: R$ ${produto.valor.toFixed(2)} `; }); alert(lista || "Nenhum produto cadastrado."); } function alterarValor() { let codigo = prompt("Digite o código do produto:"); let produto = produtos.find(p => p.codigo === codigo); if (produto) { produto.valor = Number(prompt("Digite o novo valor:")); alert("Valor alterado com sucesso!"); } else { alert("Produto não encontrado!"); } } function alterarQuantidade() { let codigo = prompt("Digite o código do produto:"); let produto = produtos.find(p => p.codigo === codigo); if (produto) { produto.quantidade = Number(prompt("Digite a nova quantidade:")); alert("Quantidade alterada com sucesso!"); } else { alert("Produto não encontrado!"); } } // Menu principal let opcao; do { opcao = prompt( "SISTEMA DE ESTOQUE\n" + "1 - Cadastrar produto\n" + "2 - Listar produtos\n" + "3 - Alterar valor\n" + "4 - Alterar quantidade\n" + "0 - Sair" ); switch (opcao) { case "1": cadastrarProduto(); break; case "2": listarProdutos(); break; case "3": alterarValor(); break; case "4": alterarQuantidade(); break; case "0": alert("Sistema encerrado!"); break; default: alert("Opção inválida!"); } } while (opcao !== "0");
