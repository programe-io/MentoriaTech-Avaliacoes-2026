import java.util.Scanner; /** * Projeto de Avaliação: Sistema de Cálculo e Verificação de Média Acadêmica * Descrição: Este programa aplica conceitos fundamentais de lógica de programação, * utilizando variáveis, operadores aritméticos, operadores relacionais e estruturas condicionais. */ public class CalculadoraMedia { public static void main(String[] args) { // Inicialização do leitor para entrada de dados via terminal Scanner scanner = new Scanner(System.in); System.out.println("=== SISTEMA DE AVALIAÇÃO ACADÊMICA ==="); // Entrada de dados: Coletando as notas do aluno System.out.print("Digite a primeira nota: "); double nota1 = scanner.nextDouble(); System.out.print("Digite a segunda nota: "); double nota2 = scanner.nextDouble(); System.out.print("Digite a terceira nota: "); double nota3 = scanner.nextDouble(); // Processamento: Cálculo da média aritmética (Operadores aritméticos) double media = (nota1 + nota2 + nota3) / 3.0; // Exibição formatada da média calculada System.out.printf("\nMédia Final Calculada: %.2f\n", media); // Estrutura Condicional (Seleção): Verificação da situação do aluno if (media >= 7.0) { System.out.println("Status: APROVADO! Parabéns pelo desempenho."); } else if (media >= 5.0 && media < 7.0) { System.out.println("Status: EM RECUPERAÇÃO. É necessário realizar a prova final."); } else { System.out.println("Status: REPROVADO. Dedique-se mais no próximo período."); } // Fechamento do recurso de leitura scanner.close(); } }
